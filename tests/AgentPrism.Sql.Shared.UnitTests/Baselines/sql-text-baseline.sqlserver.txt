UpsertWorkflow	UPDATE agentprism.workflows WITH (UPDLOCK, SERIALIZABLE)\n   SET version    = version + 1,\n       definition = @definition,\n       updated_at = @now\n OUTPUT inserted.version, inserted.updated_at\n WHERE tenant_id = @tenant_id AND name = @name;\n\nIF @@ROWCOUNT = 0\nINSERT INTO agentprism.workflows (id, tenant_id, name, version, definition, created_at, updated_at)\nOUTPUT inserted.version, inserted.updated_at\nVALUES (@id, @tenant_id, @name, 1, @definition, @now, @now);
